---
title: zod
description: More robust run-time type guards
---

The better/safer way.

```ts
import { z } from "zod";

const person_schema = z.object({
  name: z.string(),
  age: z.number(),
  birth_place: z.string().optional(),
  eyes: z.object({
    color: z.string(),
    size: z.enum(["sm", "md", "lg"]),
  }),
});

type Person = z.infer<typeof person_schema>;

function getPerson() {
  try {
    const res = await fetch("/api/person/12345");
    const person_json = await res.json();

    // `.parse` will check that the response is *actually* in the shame of `Person`
    return person_schema.parse(person_json);
  } catch (err) {
    throw err;
  }
}
```
