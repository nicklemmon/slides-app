---
title: Type guards
description: Functions that prevent runtime type errors
---

Let's write a type guard to catch the error earlier without needing to hit the server.

```ts
type ReferralSource = "facebook" | "reddit" | "google";

/** This is our type guard function */
function isValidReferralSource(arg: any): arg is ReferralSource {
  return arg === "facebook" || (arg === "reddit") | (arg === "google");
}

// The user has an invalid value in the URL 😥
const params = "?referral_source=potato";
const referral_source = new URLSearchParams(params).get("referral_source");

// We check the validity of the param here and we can show a more user-friendly error
if (!isValidReferralSource(referral_source)) {
  throw new Error(
    `Error: referral_source ${referral_source} in the URL is not valid`
  );
}

// And we save it to a database somewhere (but only when its valid) ✅
await saveReferralSource(referral_source);
```
